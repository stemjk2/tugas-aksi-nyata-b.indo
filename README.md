<script>
(() => {
  "use strict";
  const KEY = "miloDinoUltraFinalStorageV4";
  let selectedRating = 0, toastTimer, memory = [];
  const $ = id => document.getElementById(id);
  const form = $("ratingForm"), nameInput = $("name"), opinionInput = $("opinion"), submitBtn = $("submitBtn");
  const stars = [...document.querySelectorAll(".star")];
  const ratingWords = {1:"😕 Kurang suka",2:"🙂 Boleh juga",3:"😊 Lumayan enak",4:"😍 Enak banget!",5:"🤎 Sempurna! Choco lover!"};
  const EMPTY = `<div class="empty"><span>🦕</span>Belum ada ulasan.<br>Jadilah yang pertama mereview Milo Dino ini!</div>`;

  function loadRatings() {
    try {
      const data = JSON.parse(localStorage.getItem(KEY) || "[]");
      if (!Array.isArray(data)) return memory;
      return data.filter(i => i && typeof i.name === "string" && typeof i.opinion === "string" &&
        Number.isInteger(i.rating) && i.rating >= 1 && i.rating <= 5)
        .map(i => ({...i, createdAt: Number(i.createdAt) || 0}));
    } catch { return memory; }
  }
  function saveRatings(data) {
    memory = data;
    try { localStorage.setItem(KEY, JSON.stringify(data)); }
    catch { showToast("Penyimpanan browser tidak tersedia, ulasan hanya tersimpan sementara."); }
    return true;
  }
  function setRating(v) {
    selectedRating = v;
    stars.forEach((s, i) => { s.classList.toggle("active", i < v); s.setAttribute("aria-pressed", String(i + 1 === v)); });
    $("ratingLabel").textContent = ratingWords[v] || "Pilih bintang 1 sampai 5";
    validateForm();
  }
  stars.forEach(s => s.addEventListener("click", () => setRating(Number(s.dataset.value))));
  function validateForm() {
    submitBtn.disabled = !(nameInput.value.trim().length > 0 && opinionInput.value.trim().length >= 5 && selectedRating > 0);
  }
  nameInput.addEventListener("input", validateForm);
  opinionInput.addEventListener("input", validateForm);
  const escapeHTML = v => v.replace(/[&<>"']/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"})[c]);

  function render() {
    const data = loadRatings(), total = data.length;
    const sum = data.reduce((n, i) => n + i.rating, 0);
    $("totalCount").textContent = total.toLocaleString("id-ID");
    $("averageRating").textContent = total ? (sum / total).toFixed(1) : "—";
    $("fiveStarCount").textContent = data.filter(i => i.rating === 5).length.toLocaleString("id-ID");
    if (!total) { $("reviewList").innerHTML = EMPTY; return; }
    data.sort((a, b) => b.rating - a.rating || b.createdAt - a.createdAt);
    $("reviewList").innerHTML = data.map((i, idx) => `
      <article class="review">
        <div class="review-head">
          <div class="review-name">${idx === 0 && total > 1 ? "🏆 " : ""}${escapeHTML(i.name)}</div>
          <div class="review-stars">${"★".repeat(i.rating)}${"☆".repeat(5 - i.rating)}</div>
        </div>
        <p>${escapeHTML(i.opinion).replace(/\n/g, "<br>")}</p>
        <div class="review-date">${escapeHTML(i.date || "")}</div>
      </article>`).join("");
  }
  function showToast(m) {
    const t = $("toast"); t.textContent = m; t.classList.add("show");
    clearTimeout(toastTimer); toastTimer = setTimeout(() => t.classList.remove("show"), 3000);
  }
  form.addEventListener("submit", e => {
    e.preventDefault();
    const name = nameInput.value.trim(), opinion = opinionInput.value.trim();
    if (!name || opinion.length < 5 || !selectedRating) { showToast("Lengkapi nama, rating bintang, dan ulasan dulu ya!"); return; }
    const data = loadRatings();
    data.push({
      name: name.slice(0, 35), opinion: opinion.slice(0, 500), rating: selectedRating, createdAt: Date.now(),
      date: new Date().toLocaleDateString("id-ID", {day:"numeric", month:"short", year:"numeric"})
    });
